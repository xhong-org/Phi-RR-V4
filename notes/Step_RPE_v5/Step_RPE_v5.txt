	_nl_count = (array_length(_nlist))
;

// 第二次遍历 (长条用)
var _nhl_lim = _nhl_lv;
for(var _i = 0; _i < (array_length(_nhl_lim)); _i ++) {
	var _inhold = (_nhl_lim[@ _i]);
	_ref_uN(_inhold); // 强制更新Hold
	with(_inhold) {
		if ((!_inhold._is_real) || (_g_e_ap)) {
			_type_judge = JUDGMENT.PREFECT;
		} else {
			_type_judge = JUDGMENT.PREFECT;
		};
		_time_delta = (_time_down - (line.selftime));
		CHART_PLAYER_JUDGELINE_NOTE_UPDATE_JUDGMENT;
	};
	if (_inhold.__last__) then array_delete(_nhl_lim, (_i --), 1);
};

// 第一次遍历 (基础判定用)
for(var _i = _nlindex; _i < _nl_count; _i ++) {
	var _inote = (_nlist[@ _i]), _iline = (_inote.line);
	var _is_nhold = ((_inote.object_index) == obj_note_hold);
	
	//
	var _tnow = (_iline.selftime),
		_tnow_s = (_iline.selfseconds),
	;
	var _tdelta = ((_inote._time_down) - _tnow),
		_tdelta_s = ((_inote._seconds_down) - _tnow_s)
	; (_inote._time_delta) = _tdelta;
	
	// Frame
	var _frame = (_inote.frame);
	if (_tdelta <= 0) {
		(_frame ++);
		(_inote.frame) = _frame;
	};
	
	// 回收相关
	if (_tdelta_s <= 0.2) {
		var _sn_ic = (_inote[$ "_is_click"]);
		
		// 自动判定
		if ((!_inote._is_real) || (_g_e_ap)) {
			if (_sn_ic) then return(false); // 过滤点击判定
			if (!sign(_tdelta)) {
				(_inote._type_judge) = JUDGMENT.PREFECT; // 设置判定类型
				with(_inote) { CHART_PLAYER_JUDGELINE_NOTE_UPDATE_JUDGMENT }; // 触发判定事件
				if (_is_nhold) then array_add(_nhl_lv, _inote);
				_nlindex = (_i + 1);
				continue;
			};
		} else {
			// 检测判定
			if (_sn_ic) {
				(_inote._type_judge) = JUDGMENT.PREFECT; // 设置判定类型
				(_inote._time_collect) = (_iline.selfmilliseconds); // 更新回收时间
				with(_inote) { CHART_PLAYER_JUDGELINE_NOTE_UPDATE_JUDGMENT }; // 触发判定事件
				if (_is_nhold) then array_add(_nhl_lv, _inote);
				_nlindex = (_i + 1);
				continue;
			};
			
			// Miss
			if (_tdelta_s < -0.16) {
				if (_frame > 4) { // 帧检测
					(_inote._type_judge) = JUDGMENT.MISS; // 设置判定类型
					with(_inote) { CHART_PLAYER_JUDGELINE_NOTE_UPDATE_JUDGMENT }; // 触发判定事件
					if (_is_nhold) then array_add(_nhl_lv, _inote);
					_nlindex = (_i + 1);
					continue;
				};
			};
		};
	};
	
	//
	break;
};
